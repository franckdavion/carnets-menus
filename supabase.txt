-- ---------- 1. PROFILS (un par compte) ----------
create table public.profiles (
  id          uuid primary key references auth.users(id) on delete cascade,
  pseudo      text not null check (char_length(pseudo) between 2 and 40),
  role        text not null default 'member' check (role in ('member','admin')),
  status      text not null default 'pending' check (status in ('pending','active','blocked')),
  created_at  timestamptz not null default now(),
  last_seen   timestamptz
);

-- Création automatique du profil à l'inscription (pseudo saisi dans le formulaire)
create function public.handle_new_user() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  insert into public.profiles (id, pseudo)
  values (new.id, coalesce(nullif(new.raw_user_meta_data->>'pseudo',''), split_part(new.email,'@',1)));
  return new;
end $$;
create trigger on_auth_user_created after insert on auth.users
  for each row execute function public.handle_new_user();

-- Fonctions d'aide (security definer : évitent les boucles de RLS)
create function public.is_admin() returns boolean
language sql stable security definer set search_path = public as $$
  select exists (select 1 from profiles where id = auth.uid() and role = 'admin' and status = 'active');
$$;
create function public.is_active() returns boolean
language sql stable security definer set search_path = public as $$
  select exists (select 1 from profiles where id = auth.uid() and status = 'active');
$$;

-- ---------- 2. RECETTES (base commune à tous les membres) ----------
create table public.recipes (
  id          uuid primary key default gen_random_uuid(),
  legacy_id   text unique,                       -- identifiant de l'appli actuelle (r-…, r2-…) pour la migration
  author_id   uuid references public.profiles(id) on delete set null,
  name        text not null,
  cat         text not null default 'Plat',
  cats        text[] not null default '{}',
  time        int  not null default 0,
  serves      int  not null default 4,
  ing         jsonb not null default '[]',       -- [{q,u,n,r}]
  steps       text not null default '',
  image       text,                              -- photo du site / photo choisie (URL)
  photo_path  text,                              -- photo personnelle dans le stockage (bucket recipe-photos)
  tags        text[] not null default '{}',
  source      text,
  created_at  timestamptz not null default now(),
  updated_at  timestamptz not null default now()
);
create index recipes_name_idx on public.recipes (lower(name));

-- Vue pour afficher « Ajoutée par [Pseudo] » sans exposer le reste du profil
create view public.recipes_with_author with (security_invoker = true) as
  select r.*, p.pseudo as author_pseudo
  from public.recipes r left join public.profiles p on p.id = r.author_id;

-- ---------- 3. ESPACES PRIVÉS (menus, courses, favoris, vins, réglages) ----------
create table public.user_state (
  user_id     uuid primary key references public.profiles(id) on delete cascade,
  menu        jsonb not null default '{}',       -- semaines, courses, cave, préférences… (l'objet « menu » actuel)
  updated_at  timestamptz not null default now()
);
create table public.favorites (
  user_id     uuid references public.profiles(id) on delete cascade,
  recipe_id   uuid references public.recipes(id) on delete cascade,
  created_at  timestamptz not null default now(),
  primary key (user_id, recipe_id)
);
create table public.ratings (
  user_id     uuid references public.profiles(id) on delete cascade,
  recipe_id   uuid references public.recipes(id) on delete cascade,
  stars       int check (stars between 1 and 5),
  tags        text[] not null default '{}',
  at          timestamptz not null default now(),
  primary key (user_id, recipe_id, at)
);

-- ---------- 4. SÉCURITÉ (Row Level Security) ----------
alter table public.profiles   enable row level security;
alter table public.recipes    enable row level security;
alter table public.user_state enable row level security;
alter table public.favorites  enable row level security;
alter table public.ratings    enable row level security;

-- Profils : chacun voit le sien ; les membres actifs voient les pseudos (pour « Ajoutée par ») ; l'admin voit tout
create policy "profil: lire" on public.profiles for select
  using (id = auth.uid() or public.is_active());
-- Chacun peut changer SON pseudo uniquement (pas son rôle ni son statut)
create policy "profil: modifier le sien" on public.profiles for update
  using (id = auth.uid()) with check (id = auth.uid());
revoke update on public.profiles from authenticated;
grant  update (pseudo, last_seen) on public.profiles to authenticated;
-- L'admin gère les accès (rôle, statut) via une fonction dédiée
create function public.admin_set_member(member uuid, new_status text, new_role text default null)
returns void language plpgsql security definer set search_path = public as $$
begin
  if not public.is_admin() then raise exception 'réservé à l''administrateur'; end if;
  if member = auth.uid() and (new_status <> 'active' or coalesce(new_role,'admin') <> 'admin') then
    raise exception 'vous ne pouvez pas vous retirer vos propres droits';
  end if;
  update profiles set status = new_status, role = coalesce(new_role, role) where id = member;
end $$;
-- Liste des membres pour l'admin (pseudo, e-mail, statut, dates) — sans aucune donnée personnelle de menus
create function public.admin_members()
returns table (id uuid, pseudo text, email text, role text, status text, created_at timestamptz, last_seen timestamptz, recipes_count bigint)
language sql stable security definer set search_path = public as $$
  select p.id, p.pseudo, u.email, p.role, p.status, p.created_at, p.last_seen,
         (select count(*) from recipes r where r.author_id = p.id)
  from profiles p join auth.users u on u.id = p.id
  where public.is_admin()
  order by p.created_at desc;
$$;

-- Recettes : tous les membres actifs lisent ; on ajoute en son nom ; l'auteur (ou l'admin) modifie/supprime
create policy "recettes: lire"     on public.recipes for select using (public.is_active());
create policy "recettes: ajouter"  on public.recipes for insert with check (public.is_active() and author_id = auth.uid());
create policy "recettes: modifier" on public.recipes for update using (author_id = auth.uid() or public.is_admin())
                                                               with check (author_id = auth.uid() or public.is_admin());
create policy "recettes: supprimer" on public.recipes for delete using (author_id = auth.uid() or public.is_admin());

-- Espaces privés : UNIQUEMENT le propriétaire. Aucune règle pour l'admin → il ne peut ni lire ni modifier.
create policy "état privé" on public.user_state for all using (user_id = auth.uid()) with check (user_id = auth.uid());
create policy "favoris privés" on public.favorites for all using (user_id = auth.uid()) with check (user_id = auth.uid());
create policy "notes privées" on public.ratings  for all using (user_id = auth.uid()) with check (user_id = auth.uid());

-- ---------- 5. PHOTOS (stockage) ----------
insert into storage.buckets (id, name, public) values ('recipe-photos','recipe-photos', true)
  on conflict do nothing;
-- Lecture publique des photos de plats ; envoi par les membres actifs, dans leur dossier (<id>/…)
create policy "photos: envoyer" on storage.objects for insert to authenticated
  with check (bucket_id = 'recipe-photos' and public.is_active() and (storage.foldername(name))[1] = auth.uid()::text);
create policy "photos: remplacer les siennes" on storage.objects for update to authenticated
  using (bucket_id = 'recipe-photos' and (storage.foldername(name))[1] = auth.uid()::text);
create policy "photos: supprimer les siennes" on storage.objects for delete to authenticated
  using (bucket_id = 'recipe-photos' and ((storage.foldername(name))[1] = auth.uid()::text or public.is_admin()));
